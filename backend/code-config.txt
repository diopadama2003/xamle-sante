package sn.uasz.xamle.config;

import lombok.RequiredArgsConstructor;
import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.crypto.password.PasswordEncoder;
import sn.uasz.xamle.model.Role;
import sn.uasz.xamle.model.Utilisateur;
import sn.uasz.xamle.repository.UtilisateurRepository;

@Configuration
@RequiredArgsConstructor
public class DataInitializer {

    private final UtilisateurRepository repo;
    private final PasswordEncoder encoder;

    @Bean
    CommandLineRunner initComptes() {
        return args -> {
            creerSiAbsent("admin@xamle.sn", "Admin", "Santé", "Admin@2026", Role.ADMIN_SANTE);
            creerSiAbsent("moderateur@xamle.sn", "Modérateur", "Médical", "Modo@2026", Role.MODERATEUR);
        };
    }

    private void creerSiAbsent(String email, String nom, String prenom, String mdp, Role role) {
        if (!repo.existsByEmail(email)) {
            repo.save(Utilisateur.builder()
                    .nom(nom).prenom(prenom).email(email)
                    .motDePasse(encoder.encode(mdp))
                    .role(role)
                    .build());
        }
    }
}

package sn.uasz.xamle.config;

import lombok.RequiredArgsConstructor;
import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import sn.uasz.xamle.model.Campagne;
import sn.uasz.xamle.model.TypeCampagne;
import sn.uasz.xamle.repository.CampagneRepository;
import sn.uasz.xamle.service.QuartierCatalogue;

import java.time.LocalDateTime;

@Configuration
@RequiredArgsConstructor
public class DemoCampagnesInitializer {

    private final CampagneRepository repo;
    private final QuartierCatalogue catalogue;

    @Bean
    CommandLineRunner initCampagnesDemo() {
        return args -> {
            if (repo.count() > 0) {
                return;
            }
            LocalDateTime maintenant = LocalDateTime.now();
            // Campagnes FICTIVES de démonstration. La première commence dans 20 h :
            // elle permet de tester le rappel SMS 24 h avant.
            ajouter("Démo : vaccination contre la rougeole", TypeCampagne.VACCINATION,
                    "Vaccination gratuite des enfants de 9 mois à 5 ans.",
                    "Centre de santé de démonstration", "Boucotte",
                    maintenant.plusHours(20), 4);
            ajouter("Démo : dépistage du paludisme", TypeCampagne.DEPISTAGE,
                    "Tests rapides gratuits pour toute la famille.",
                    "Place publique", "Escale",
                    maintenant.plusDays(5), 6);
            ajouter("Démo : sensibilisation à l'hygiène des mains", TypeCampagne.SENSIBILISATION,
                    "Séance d'information et démonstration pour les écoles du quartier.",
                    "École de démonstration", "Lyndiane",
                    maintenant.plusDays(12), 3);
        };
    }

    private void ajouter(String titre, TypeCampagne type, String description,
                         String lieu, String quartier, LocalDateTime debut, int dureeHeures) {
        QuartierCatalogue.Quartier q = catalogue.trouver(quartier).orElseThrow();
        repo.save(Campagne.builder()
                .titre(titre).type(type).description(description)
                .lieu(lieu).quartier(q.nom())
                .latitude(q.latitude()).longitude(q.longitude())
                .dateDebut(debut).dateFin(debut.plusHours(dureeHeures))
                .build());
    }
}

package sn.uasz.xamle.config;

import lombok.RequiredArgsConstructor;
import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import sn.uasz.xamle.model.Signalement;
import sn.uasz.xamle.model.StatutValidation;
import sn.uasz.xamle.model.Symptome;
import sn.uasz.xamle.repository.SignalementRepository;

import java.time.LocalDateTime;

@Configuration
@RequiredArgsConstructor
public class DemoSignalementsInitializer {

    private final SignalementRepository repo;

    @Bean
    CommandLineRunner initSignalementsDemo() {
        return args -> {
            if (repo.count() > 0) {
                return;
            }
            // Données FICTIVES de démonstration
            for (int i = 0; i < 7; i++) ajouter(Symptome.FIEVRE, "Escale", StatutValidation.VALIDE, i % 5 + 1);
            for (int i = 0; i < 4; i++) ajouter(Symptome.DIARRHEE, "Boucotte", StatutValidation.VALIDE, i + 1);
            for (int i = 0; i < 3; i++) ajouter(Symptome.TOUX, "Lyndiane", StatutValidation.VALIDE, i + 2);
            // Seulement 2 : sous le seuil d'anonymat, ce quartier reste invisible
            for (int i = 0; i < 2; i++) ajouter(Symptome.FIEVRE, "Néma", StatutValidation.VALIDE, i + 1);
            // En attente de modération
            ajouter(Symptome.VOMISSEMENTS, "Santhiaba", StatutValidation.EN_ATTENTE, 0);
            ajouter(Symptome.FIEVRE, "Kénia", StatutValidation.EN_ATTENTE, 0);
        };
    }

    private void ajouter(Symptome symptome, String quartier, StatutValidation statut, int joursPasses) {
        repo.save(Signalement.builder()
                .symptome(symptome)
                .quartier(quartier)
                .statut(statut)
                .dateCreation(LocalDateTime.now().minusDays(joursPasses))
                .build());
    }
}

package sn.uasz.xamle.config;

import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.http.HttpStatus;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.HttpStatusEntryPoint;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import sn.uasz.xamle.security.JwtAuthFilter;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
@RequiredArgsConstructor
public class SecurityConfig {

    private final JwtAuthFilter jwtAuthFilter;

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
                .csrf(csrf -> csrf.disable())
                .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .exceptionHandling(e -> e.authenticationEntryPoint(
                        new HttpStatusEntryPoint(HttpStatus.UNAUTHORIZED)))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers(
                                "/swagger-ui.html",
                                "/swagger-ui/**",
                                "/v3/api-docs/**",
                                "/api/health",
                                "/api/auth/**",
                                "/api/sos/**",
                                "/api/signalements/**",
                                "/api/campagnes/**",
                                "/error"
                        ).permitAll()
                        .requestMatchers(HttpMethod.GET,
                                "/api/specialistes/**",
                                "/api/etablissements/**").permitAll()
                        .requestMatchers("/api/admin/**").hasRole("ADMIN_SANTE")
                        .requestMatchers("/api/moderation/**").hasAnyRole("MODERATEUR", "ADMIN_SANTE")
                        .anyRequest().authenticated()
                )
                .addFilterBefore(jwtAuthFilter, UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }
}
